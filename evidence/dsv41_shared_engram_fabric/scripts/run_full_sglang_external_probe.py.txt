def selected_rows(rows: int, sizes: tuple[int, ...]) -> list[int]:
    half = sum(sizes[: len(sizes) // 2])
    return sorted({0, sizes[0] - 1, sizes[0], half - 1, half, rows // 2, rows - 1})


def reference(
    model: Path, index: dict, layer: int, positions: list[int]
) -> torch.Tensor:
    weight_key = f"layers.{layer}.engram.embed.weight"
    scale_key = f"layers.{layer}.engram.embed.scale"
    with safe_open(model / index[weight_key], framework="pt", device="cpu") as source:
        weight = source.get_slice(weight_key)
        values = torch.cat([weight[row : row + 1] for row in positions])
    with safe_open(model / index[scale_key], framework="pt", device="cpu") as source:
        scale = source.get_slice(scale_key)
        exponents = torch.cat([scale[row : row + 1] for row in positions])
    assert values.dtype == torch.float8_e4m3fn
    assert exponents.dtype == torch.float8_e8m0fnu
    block = 32
    raw_exponents = exponents.view(torch.uint8).to(torch.int32)
    factors = (raw_exponents << 23).view(torch.float32)
    factors = torch.where(raw_exponents == 0, 2.0**-127, factors)
    return (
        values.to(torch.float32).unflatten(-1, (-1, block))
        * factors.unsqueeze(-1)
    ).flatten(-2).to(torch.bfloat16)


def consumer(endpoint: str, token_file: Path, generation: str, output: Path) -> None:
    from sglang.kernels.ops.embeddings.engram_gather import engram_gather
    from sglang.srt.layers.engram import EngramLayout
    from sglang.srt.layers.engram_external import attach_table, close_attached_tables

    torch.cuda.set_device(1)
    config = json.loads((MODEL / "config.json").read_text())
    text_config = config.get("text_config", config)
    layout = EngramLayout.from_config(SimpleNamespace(**text_config))
    assert layout is not None
    index = json.loads((MODEL / "model.safetensors.index.json").read_text())[
        "weight_map"
    ]
    reports = []
    try:
        for layer_index, layer in enumerate(layout.layer_ids):
            rows = layout.num_embeddings[layer_index]
            dim = layout.head_dim
            bounds = layout.column_bounds(layer_index)
            sizes = tuple(end - start for start, end in zip(bounds, bounds[1:]))
            positions = selected_rows(rows, sizes)
            rss_before = psutil.Process().memory_info().rss
            table = attach_table(
                endpoint=endpoint,
                token_file=token_file,
                generation=generation,
                checkpoint_dir=MODEL,
                layer=layer,
                rows=rows,
                dim=dim,
                head_sizes=sizes,
                device=1,
            )
            rss_delta = psutil.Process().memory_info().rss - rss_before
            expected = reference(MODEL, index, layer, positions)
            ids = torch.tensor(positions, dtype=torch.int32, device="cuda")
            gathered = torch.empty(
                (len(positions), dim), dtype=torch.bfloat16, device="cuda"
            )
            engram_gather(
                table.weight_ptr, table.scale_ptr, ids, gathered, dim, 32
            )
            torch.cuda.synchronize()
            torch.testing.assert_close(gathered.cpu(), expected, rtol=0, atol=0)
            graph = torch.cuda.CUDAGraph()
            with torch.cuda.graph(graph):
                engram_gather(
                    table.weight_ptr, table.scale_ptr, ids, gathered, dim, 32
                )
            graph.replay()
            torch.cuda.synchronize()
            torch.testing.assert_close(gathered.cpu(), expected, rtol=0, atol=0)
            ids.copy_(ids.flip(0))
            graph.replay()
            torch.cuda.synchronize()
            torch.testing.assert_close(
                gathered.cpu(), expected.flip(0), rtol=0, atol=0
            )
            del graph
            reports.append(
                {
                    "layer": layer,
                    "rows": rows,
                    "positions": positions,
                    "table_bytes": table.mapping.table.size_bytes,
                    "rss_delta_at_attach": rss_delta,
                    "expected_bf16_sha256": hashlib.sha256(
                        expected.view(torch.uint8).numpy().tobytes()
                    ).hexdigest(),
                    "changed_ids_graph_replay": True,
                }
            )
            close_attached_tables()
        output.write_text(json.dumps({"result": "pass", "tables": reports}, indent=2))
        print(json.dumps({"result": "pass", "layers": [x["layer"] for x in reports]}))
    finally:
        close_attached_tables()


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--consumer", action="store_true")
    parser.add_argument("--endpoint")
    parser.add_argument("--token-file", type=Path)
    parser.add_argument("--generation")
    parser.add_argument("--output", type=Path)
    args = parser.parse_args()
    if args.consumer:
        consumer(args.endpoint, args.token_file, args.generation, args.output)
        return
    ARTIFACTS.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory() as temporary:
        token_file = Path(temporary) / "token"
        token_file.write_text(secrets.token_hex(32))
        os.chmod(token_file, 0o600)
        with socket.socket() as probe:
            probe.bind(("127.0.0.1", 0))
            port = probe.getsockname()[1]
        environment = os.environ.copy()
        environment["PYTHONPATH"] = str(VLLM_SOURCE)
        ready = threading.Event()
        lines: list[str] = []
        with (ARTIFACTS / "full_owner.log").open("w") as log:
            owner = subprocess.Popen(
                [
                    str(VLLM_PYTHON),
                    "-m",
                    "vllm.models.deepseek_v41.nvidia.ops.shared_engram_owner",
                    "--model-dir",
                    str(MODEL),
                    "--listen-host",
                    "127.0.0.1",
                    "--listen-port",
                    str(port),
                    "--token-file",
                    str(token_file),
                    "--device",
                    "0",
                ],
                env=environment,
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                text=True,
                start_new_session=True,
            )

            def read_owner() -> None:
                assert owner.stdout is not None
                for line in owner.stdout:
                    lines.append(line)
                    log.write(line)
                    log.flush()
                    if "ENGRAM_OWNER_READY" in line:
                        ready.set()
                ready.set()

            thread = threading.Thread(target=read_owner)
            thread.start()
            try:
                if not ready.wait(timeout=1200):
                    raise RuntimeError("owner did not publish: " + "".join(lines[-20:]))
                ready_line = next(
                    (line for line in lines if "ENGRAM_OWNER_READY" in line), None
                )
                if ready_line is None:
                    raise RuntimeError("owner exited before ready: " + "".join(lines[-20:]))
                generation = ready_line.split("generation=", 1)[1].strip()
                result = subprocess.run(
                    [
                        sys.executable,
                        str(Path(__file__).resolve()),
                        "--consumer",
                        "--endpoint",
                        f"127.0.0.1:{port}",
                        "--token-file",
                        str(token_file),
                        "--generation",
                        generation,
                        "--output",
                        str(ARTIFACTS / "full_probe.json"),
                    ],
                    capture_output=True,
                    text=True,
                    timeout=900,
                    env={
                        **os.environ,
                        "PYTHONPATH": str(SGLANG_SOURCE / "python"),
                        "CUDA_VISIBLE_DEVICES": "0,1",
                    },
                )
                (ARTIFACTS / "full_consumer.log").write_text(
                    result.stdout + result.stderr
                )
                if result.returncode:
                    raise RuntimeError(
                        f"consumer exit code {result.returncode}: "
                        + "\n".join((result.stdout + result.stderr).splitlines()[-25:])
                    )
                print(result.stdout.strip(), flush=True)
            finally:
                with suppress(ProcessLookupError):
                    os.killpg(owner.pid, signal.SIGTERM)
                try:
                    owner.wait(timeout=120)
                except subprocess.TimeoutExpired:
                    os.killpg(owner.pid, signal.SIGKILL)
                    owner.wait()
                thread.join(timeout=10)
            if owner.returncode:
                raise RuntimeError(f"owner exit code {owner.returncode}")
            print(json.dumps({"owner_exit_code": owner.returncode}), flush=True)


if __name__ == "__main__":
    main()
